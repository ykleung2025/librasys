function doGet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const booksSheet = ss.getSheetByName("Books"); const studentsSheet = ss.getSheetByName("Students"); const recordsSheet = ss.getSheetByName("Records"); const books = booksSheet ? booksSheet.getDataRange().getValues().slice(1) : []; const students = studentsSheet ? studentsSheet.getDataRange().getValues().slice(1) : []; const records = recordsSheet ? recordsSheet.getDataRange().getValues().slice(1) : []; const result = { status: "success", books: books.map(r => ({ id: r[0], title: r[1], status: r[2] })), students: students.map(r => ({ id: r[0], name: r[1] })), records: records.map(r => ({ id: r[0], date: r[1] ? new Date(r[1]).toLocaleString() : '', bookId: r[2], studentId: r[3], photoUrl: r[4], status: r[5] })) }; return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const data = JSON.parse(e.postData.contents); if (data.action === 'uploadPhoto') { // 借用人：拍照上傳相片存至 Drive 並紀錄至 Google Sheet const recordsSheet = ss.getSheetByName("Records"); let photoUrl = "無照片"; if (data.imageBase64) { const folder = getOrCreateFolder("圖書借閱照片庫"); const blob = Utilities.newBlob(Utilities.base64Decode(data.imageBase64.split(',')[1]), 'image/jpeg', `borrow_${Date.now()}.jpg`); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); photoUrl = file.getUrl(); } const newId = 'REC' + Date.now(); recordsSheet.appendRow([newId, new Date(), '', '', photoUrl, '待管理員核對']); return respondJSON({ status: 'success', message: '照片上傳成功！請等待管理員核對借出。' }); } if (data.action === 'issueBook') { // 管理員：選取/輸入 書編號 與 學生編號，辦理借出 const booksSheet = ss.getSheetByName("Books"); const recordsSheet = ss.getSheetByName("Records"); const booksData = booksSheet.getDataRange().getValues(); let bookFound = false; // 1. 更新 Books 表格狀態為 已借出 for (let i = 1; i < booksData.length; i++) { if (booksData[i][0].toString().trim() === data.bookId.trim()) { booksSheet.getRange(i + 1, 3).setValue('已借出'); bookFound = true; break; } } if (!bookFound) { return respondJSON({ status: 'error', message: '找不到該書籍編號！' }); } // 2. 寫入 Records 借閱紀錄 const newId = 'REC' + Date.now(); recordsSheet.appendRow([newId, new Date(), data.bookId, data.studentId, data.photoUrl || '手動登記', '已借出']); return respondJSON({ status: 'success', message: `成功辦理借出！書號：${data.bookId}，學生：${data.studentId}` }); } if (data.action === 'returnBook') { // 管理員：還書 const booksSheet = ss.getSheetByName("Books"); const booksData = booksSheet.getDataRange().getValues(); for (let i = 1; i < booksData.length; i++) { if (booksData[i][0].toString().trim() === data.bookId.trim()) { booksSheet.getRange(i + 1, 3).setValue('未借出'); break; } } return respondJSON({ status: 'success', message: '已完成還書手續！' }); } } function getOrCreateFolder(folderName) { const folders = DriveApp.getFoldersByName(folderName); if (folders.hasNext()) return folders.next(); return DriveApp.createFolder(folderName); } function respondJSON(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
